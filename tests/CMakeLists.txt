cmake_minimum_required(VERSION 3.16)
project(funiq_strategy_example_test LANGUAGES CXX)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
add_executable(stop_profit_loss_strategy_test stop_profit_loss_strategy_test.cpp)
target_include_directories(stop_profit_loss_strategy_test PRIVATE ../include)
target_link_libraries(stop_profit_loss_strategy_test PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
target_compile_features(stop_profit_loss_strategy_test PRIVATE cxx_std_17)
# The example checks must also execute in Release builds.
if(MSVC)
    target_compile_options(stop_profit_loss_strategy_test PRIVATE /UNDEBUG)
else()
    target_compile_options(stop_profit_loss_strategy_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME paper_stop_profit_loss COMMAND stop_profit_loss_strategy_test)
